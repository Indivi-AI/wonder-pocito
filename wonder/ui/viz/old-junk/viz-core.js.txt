import { dsls, jb, coreUtils } from '@jb6/core'
import '@jb6/react'

const { react: { ReactComp, 'react-comp': { comp }, 'react-metadata': { importUrl } } } = dsls

// ============================================================================
// jb.vizUtils — shared ECharts kit for every inline widget. Charts render with
// the SVG renderer (renderer:'svg') so they serialize in jsdom for reactTests and
// look identical in the browser. Each widget supplies only a pure
// optionFrom(props) -> echartsOption; vizComp() owns init/resize/dispose and the
// one-time text-measure shim that lets ECharts run without a canvas in jsdom.
// ============================================================================

const ECHARTS_IMPORT = '@jb6/react/lib/echarts/echarts.mjs'   // vendored next to codemirror; resolves in node + browser via the @jb6/react import map

// theme: brand override point. A brand file (e.g. comax-brand.js) Object.assigns
// {palette, accent, fontFamily} onto jb.vizTheme; values are read at render time,
// so brand and kit can load in any order.
const THEME = jb.vizTheme = Object.assign({
  palette: ['#2563eb', '#16a34a', '#f59e0b', '#dc2626', '#7c3aed', '#db2777', '#0d9488', '#ea580c', '#4f46e5', '#65a30d', '#0891b2', '#9333ea'],
  accent: '#2563eb',
  fontFamily: 'ui-sans-serif, system-ui, sans-serif'
}, jb.vizTheme)
const DIM = '#cbd5e1', INK = '#1e293b', MUTE = '#64748b'
const colorAt = i => THEME.palette[((i % THEME.palette.length) + THEME.palette.length) % THEME.palette.length]

const compact = v => {
  const a = Math.abs(v)
  if (a >= 1e9) return (v / 1e9).toFixed(a >= 1e10 ? 0 : 1) + 'B'
  if (a >= 1e6) return (v / 1e6).toFixed(a >= 1e7 ? 0 : 1) + 'M'
  if (a >= 1e3) return (v / 1e3).toFixed(a >= 1e4 ? 0 : 1) + 'K'
  return String(Math.round(v * 100) / 100)
}
const fmtNum = (v, format) => {
  if (v == null || (typeof v === 'number' && isNaN(v))) return ''
  const n = +v
  if (format === '₪') return compact(n) + ' ₪'
  if (format === '%') return compact(n) + '%'
  if (format === '$') return '$' + compact(n)
  if (format === 'compact') return compact(n)
  if (format === 'int') return String(Math.round(n))
  return compact(n)
}

// --- highlight contract (unchanged vocabulary the LLM emits) -----------------
// highlight ∈ name | index | {name|index, note?} | {min|max, note?} | array of these
const asHlList = h => h == null ? [] : (Array.isArray(h) ? h : [h])
const hasHighlight = h => asHlList(h).length > 0
const resolveExtremes = (highlight, items) => asHlList(highlight).map(h => {
  if (h && typeof h === 'object' && (h.min || h.max) && items?.length) {
    const pick = items.reduce((a, b) => (h.max ? b.value > a.value : b.value < a.value) ? b : a)
    return { name: pick.name, note: h.note }
  }
  return h
})
const matchHighlight = (highlight, name, index) => asHlList(highlight).some(h => {
  if (h == null) return false
  if (typeof h === 'object') return (h.name != null && String(h.name) === String(name)) || (h.index != null && h.index === index)
  if (typeof h === 'number') return h === index
  return String(h) === String(name)
})
const highlightNote = (highlight, name) => asHlList(highlight).find(h => h && typeof h === 'object' && h.note && (h.name == null || String(h.name) === String(name)))?.note
// per-item color for bar/pie-like series given the highlight state
const itemColors = (highlight, items) => { const active = hasHighlight(highlight)
  return items.map((d, i) => active ? (matchHighlight(highlight, d.name, i) ? (d.color || colorAt(i)) : DIM) : (d.color || colorAt(i))) }

// --- common option blocks ----------------------------------------------------
const titleBlock = (title, note) => (title || note) ? { title: { text: title || '', subtext: note || '', left: 8, top: 6,
  textStyle: { fontSize: 15, fontWeight: 600, color: INK }, subtextStyle: { fontSize: 12, fontWeight: 600, color: THEME.accent } } } : {}
const legendBlock = (show, names) => show ? { legend: { data: names, bottom: 2, textStyle: { color: '#475569', fontSize: 12 } } } : {}

// responsify — width-aware pass over every option at render time, so each widget lays out at the REAL container width:
// a bottom legend that wraps gets its rows reserved in grid.bottom (4+ rows degrade to one scrollable row), crowded
// category x labels tilt + truncate instead of overlapping or vanishing, and category y labels (hbar) scale to the width.
const responsify = (option, width) => {
  const small = width < 480, legend = option.legend
  if (legend?.bottom != null && legend.type != 'scroll' && legend.data?.length) {
    small && Object.assign(legend, { itemWidth: 18, itemGap: 8, textStyle: { ...legend.textStyle, fontSize: 11 } })
    const fs = legend.textStyle?.fontSize || 12, gap = legend.itemGap ?? 10
    let rows = 1, x = 0
    legend.data.forEach(name => { const w = (legend.itemWidth || 25) + 5 + String(name).length * fs * 0.58
      x && x + w > width - 16 ? (rows++, x = w + gap) : x += w + gap })
    if (rows > 3) legend.type = 'scroll'
    else if (rows > 1 && option.grid) option.grid.bottom = Math.max(+option.grid.bottom || 0, rows * (fs + 10) + 18)
  }
  coreUtils.asArray(option.xAxis).forEach(ax => {
    if (ax?.type != 'category' || !ax.data?.length) return
    const lbl = ax.axisLabel ||= {}, per = width / ax.data.length
    small && (lbl.fontSize = Math.min(lbl.fontSize || 11, 10))
    const need = 20 + Math.max(0, ...ax.data.map(c => String(c).length)) * (lbl.fontSize || 11) * 0.58
    if (ax.data.length <= 12 && per < Math.min(110, need) && lbl.rotate == null && lbl.width == null)
      Object.assign(lbl, { rotate: per < 55 ? 45 : 30, width: Math.min(110, Math.max(56, Math.round(per * 1.8))), overflow: 'truncate', hideOverlap: false, interval: 0 })
  })
  small && coreUtils.asArray(option.yAxis).forEach(ax => {
    ax?.type == 'category' && ax.axisLabel && Object.assign(ax.axisLabel, { width: Math.min(ax.axisLabel.width || 999, Math.round(width * 0.34)), overflow: 'truncate' })
  })
  return option
}

// --- drill contract -----------------------------------------------------------
// spec.drill = {kind, title, sql, question?, label?} | [...]. A click normalizes to
// {name, series, x, y, value}; templates fill {name}.. as text and {name:q} as a
// SQL-quoted literal; drill.sql column aliases must match the target kind's shape
// (rowsToSpec below), so the side plot renders straight from the query rows.
const defaultClickInfo = p => ({ name: String(p.name || p.data?.name || p.seriesName || ''), series: p.seriesName,
  x: p.name ?? (Array.isArray(p.value) ? p.value[0] : undefined), y: Array.isArray(p.value) ? p.value[1] : p.value })
const fillDrill = (tpl, info) => String(tpl || '').replace(/\{(name|series|x|y|value)(:q)?\}/g, (m, k, q) =>
  info[k] == null ? m : q ? `'${String(info[k]).replace(/'/g, "''")}'` : String(info[k]))
const uniq = xs => [...new Set(xs)]
const groupBy = (rows, k) => rows.reduce((acc, r) => ((acc[r[k]] = acc[r[k]] || []).push(r), acc), {})
const pivot = rows => {
  const categories = uniq(rows.map(r => String(r.category ?? r.x)))
  return { categories, series: Object.entries(groupBy(rows, 'series')).map(([name, g]) =>
    ({ name, values: categories.map(c => +(g.find(r => String(r.category ?? r.x) == c)?.value) || 0) })) }
}
const rowsToSpec = (kind, rows) => {
  const rs = coreUtils.asArray(rows).filter(r => r && typeof r == 'object')
  switch (kind) {
    case 'table': return { columns: Object.keys(rs[0] || {}).map(key => ({ key, label: key })), rows: rs }
    case 'kpi': return { items: rs.map(r => ({ label: String(r.label ?? r.name ?? ''), value: r.value, delta: r.delta })) }
    case 'histogram': return { values: rs.map(r => +(r.value ?? Object.values(r)[0]) || 0) }
    case 'gauge': return { ...rs[0] }
    case 'line': case 'area': return { series: rs.some(r => r.series != null)
      ? Object.entries(groupBy(rs, 'series')).map(([name, g]) => ({ name, points: g.map(r => ({ x: r.x, y: +r.y || 0 })) }))
      : [{ name: '', points: rs.map(r => ({ x: r.x ?? r.name, y: +(r.y ?? r.value) || 0 })) }] }
    case 'scatter': case 'bubble': case 'bullet': return { data: rs }
    case 'heatmap': return { xCategories: uniq(rs.map(r => String(r.x))), yCategories: uniq(rs.map(r => String(r.y))), data: rs }
    case 'boxplot': return { data: Object.entries(groupBy(rs, 'name')).map(([name, g]) => ({ name, values: g.map(r => +r.value || 0) })) }
    case 'groupedBar': case 'stackedBar': return pivot(rs)
    case 'radar': {
      const p = pivot(rs.map(r => ({ category: r.indicator ?? r.x ?? r.name, series: r.series, value: r.value })))
      const max = Math.max(1, ...rs.map(r => +r.value || 0))
      return { indicators: p.categories.map(name => ({ name, max })), series: p.series }
    }
    default: return { data: rs.map(r => ({ name: String(r.name ?? Object.values(r)[0]), value: +(r.value ?? Object.values(r)[1]) || 0 })) }
  }
}

// --- ECharts platform shim: pure-JS text metrics so jsdom needs no canvas ----
const ensurePlatform = echarts => {
  if (jb.__echartsReady) return
  jb.__echartsReady = true
  if (!globalThis.window?.testing) return   // real browser: keep native, accurate canvas metrics
  const measure = (text, font) => ({ width: String(text).length * (parseInt(String(font).match(/(\d+)px/)?.[1] || '12', 10)) * 0.6 })
  echarts.setPlatformAPI({ measureText: measure, createCanvas: () => ({ getContext: () => ({ measureText: t => measure(t, '12px') }) }) })
}

// --- the shared widget: init SVG chart, setOption, resize, dispose -----------
// Each kind calls vizComp(kind, optionFrom, description, clickInfo?). optionFrom(props, jb.vizUtils)
// is pure; clickInfo(echartsParams, props) normalizes a click to {name, series, x, y, value}.
const vizComp = (kind, optionFrom, description, clickInfo) => ReactComp(`${kind}Widget`, {
  description: description || `Inline ${kind} chart; highlight emphasizes an element`,
  impl: comp({
    hFunc: (ctx, { react: { h, useRef, useEffect } }) => props => {
      const echarts = jb.reactRepository.importCache[ECHARTS_IMPORT]
      const { titleBadge, ...chartProps } = props
      const width = chartProps.width || 540, height = chartProps.height || 320, hasDomTitle = titleBadge && chartProps.title
      ensurePlatform(echarts)
      const ref = useRef(null)
      useEffect(() => {
        const el = ref.current
        if (!el) return
        const chart = echarts.init(el, null, { renderer: 'svg', width: el.clientWidth || width, height })   // jsdom clientWidth is 0 → keep fixed width for tests
        const render = () => {   // options lay out from the real width (pie geometry, histogram bins), so recompute on every resize
          chart.resize({ width: el.clientWidth || width })
          chart.setOption({ color: THEME.palette, animation: !globalThis.window?.testing, textStyle: { fontFamily: THEME.fontFamily },
            ...responsify(optionFrom({ ...chartProps, width: el.clientWidth || width, ...(hasDomTitle ? { title: '' } : {}) }, jb.vizUtils), el.clientWidth || width) },
            { notMerge: true })   // options are complete per render; merge would keep narrow-width adaptations (rotate/scroll) after growing wide
        }
        render()
        chartProps.onEvent && chart.on('click', p => chartProps.onEvent({ type: 'drill', ...(clickInfo || defaultClickInfo)(p, chartProps), spec: chartProps }))
        const ro = globalThis.ResizeObserver && new globalThis.ResizeObserver(render)
        ro?.observe(el)
        return () => { ro?.disconnect(); chart.dispose() }
      }, [JSON.stringify(chartProps)])
      // direction ltr: ECharts computes text geometry assuming LTR; an inherited rtl (Hebrew host pages)
      // pushes left-anchored titles/labels off-canvas. Hebrew strings still render correctly via unicode bidi.
      return h('div:viz-widget', { style: { direction: 'ltr', width: '100%', maxWidth: width + 'px', height: (height + (hasDomTitle ? 34 : 0)) + 'px',
          background: '#fff', border: '1px solid #f1f5f9', borderRadius: '14px', boxShadow: '0 1px 3px rgba(0,0,0,0.06)', boxSizing: 'border-box' } },
        hasDomTitle && h('div', { style: { direction: 'rtl', display: 'flex', alignItems: 'center', justifyContent: 'flex-start', gap: '6px',
          fontSize: '15px', fontWeight: 600, color: INK, padding: '12px 14px 0', marginBottom: '4px', boxSizing: 'border-box' } }, titleBadge, chartProps.title),
        h('div:viz-chart', { ref, style: { direction: 'ltr', width: '100%', height: height + 'px' } }))
    },
    metadata: [importUrl(() => ECHARTS_IMPORT)]
  })
})

jb.vizUtils = {
  get PALETTE() { return THEME.palette }, DIM, INK, MUTE, colorAt, compact, fmtNum,
  hasHighlight, resolveExtremes, matchHighlight, highlightNote, itemColors,
  titleBlock, legendBlock, vizComp, asArray: coreUtils.asArray, ECHARTS_IMPORT,
  fillDrill, rowsToSpec
}

// ============================================================================
// VizWidget — universal host. Dispatches a {kind,...} spec to `${kind}Widget`.
// When spec.drill exists, a click runs its SQL via the runSql prop and renders
// the side plot in a panel below; drill.question escalates through onEvent as
// {type:'question', question}. Without drill, clicks pass through to onEvent.
// ============================================================================
ReactComp('VizWidget', {
  description: 'Inline data-visualization host: dispatches a {kind,...} spec to the matching chart widget; spec.drill + runSql opens a click-driven side plot',
  impl: comp({
    hFunc: (ctx, { react: { h, hh, useState } }) => ({ spec, onEvent, runSql }) => {
      const s = spec || {}
      const widget = dsls.react['react-comp'][`${s.kind}Widget`]
      const [panel, setPanel] = useState(null)
      if (!widget) return h('div', { style: { color: '#dc2626', fontSize: '12px', padding: '8px' } }, `viz: unknown kind "${s.kind}"`)
      const drills = coreUtils.asArray(s.drill).filter(d => d && (d.sql || d.question))
      const openDrill = async (d, info) => {
        const base = { d, info, title: fillDrill(d.title || info.name, info), question: d.question && fillDrill(d.question, info) }
        if (!(d.sql && runSql)) return base.question && onEvent?.({ type: 'question', question: base.question, ...info, spec: s })
        setPanel({ ...base, loading: true })
        try {
          const rows = await runSql(fillDrill(d.sql, info))
          setPanel({ ...base, childSpec: { kind: d.kind, title: base.title, valueFormat: d.valueFormat, ...rowsToSpec(d.kind, rows) } })
        } catch (e) { setPanel({ ...base, error: String(e?.message || e) }) }
      }
      const handleEvent = e => drills.length ? openDrill(drills[0], e) : onEvent?.(e)
      const btn = { border: '1px solid #e2e8f0', borderRadius: '999px', background: '#fff', color: '#475569', fontSize: '12px', padding: '4px 10px', cursor: 'pointer' }
      return h('div', { style: { display: 'flex', flexDirection: 'column', gap: '10px' } },
        hh(ctx, widget, { ...s, onEvent: (drills.length || onEvent) && handleEvent }),
        panel && h('div:viz-drill-panel', { style: { border: '1px solid #e2e8f0', borderRadius: '14px', padding: '12px', background: '#fafbfc',
            width: '100%', maxWidth: (s.width || 540) + 'px', boxSizing: 'border-box' } },
          h('div', { style: { display: 'flex', alignItems: 'center', gap: '8px', flexWrap: 'wrap', marginBottom: '8px' } },
            h('div', { style: { fontSize: '13px', fontWeight: 600, color: INK, flex: 1 } }, panel.title),
            drills.length > 1 && drills.map((d, i) =>
              h('button', { key: i, style: { ...btn, fontWeight: d === panel.d ? 700 : 400 }, onClick: () => openDrill(d, panel.info) }, d.label || d.kind)),
            panel.question && onEvent &&
              h('button', { style: btn, onClick: () => onEvent({ type: 'question', question: panel.question, ...panel.info, spec: s }) }, 'המשך בצ׳אט'),
            h('button', { style: btn, 'aria-label': 'close drill', onClick: () => setPanel(null) }, '✕')),
          panel.loading && h('div', { style: { fontSize: '12px', color: MUTE } }, '…'),
          panel.error && h('div', { style: { fontSize: '12px', color: '#dc2626' } }, panel.error),
          panel.childSpec && hh(ctx, dsls.react['react-comp'].VizWidget, { spec: panel.childSpec, runSql })))
    }
  })
})
