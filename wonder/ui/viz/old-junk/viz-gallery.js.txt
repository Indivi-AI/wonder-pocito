ReactComp('viz.Gallery', {
  params: [
    {id: 'visualizations', type: 'react-comp<react>[]'}
  ],
  impl: comp({
    hFunc: (ctx, { react: { h } }, { visualizations }) => () => h('div', {
      style: { background: '#f8fafc', padding: '20px', minHeight: '100vh', fontFamily: 'ui-sans-serif, system-ui' }
    },
    h('h1', { style: { fontSize: '20px', fontWeight: 700, color: '#0f172a', marginBottom: '16px' } }, 'Visualization gallery'),
    h('div', { style: { display: 'flex', flexWrap: 'wrap', gap: '16px', alignItems: 'flex-start' } },
      visualizations.map((visualization, i) => h('div', {
        key: i, style: { flex: '1 1 320px', minWidth: 0, maxWidth: '620px' }
      }, h(visualization)))))
  })
})

Test('reactTest.viz.namespacedPie', {
  impl: reactTest({
    testedComp: viz.Pie('TGP pie', [
      {name: 'A', value: 2},
      {name: 'B', value: 1}
    ], { highlight: 'A' }),
    expectedResult: and(contains('TGP pie'), contains('A')),
    userActions: waitForText('TGP pie')
  })
})
