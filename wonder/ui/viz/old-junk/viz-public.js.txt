import { dsls } from '@jb6/core'

const { tgp: { DefComponents }, react: { ReactComp, 'react-comp': { comp } } } = dsls

DefComponents('Pie,Bar,HBar,Scatter,Bubble,Boxplot,Heatmap,Funnel,Treemap,Waterfall,Bullet', templateParam =>
  ReactComp(`viz.${templateParam}`, {
    params: [
      {id: 'title', as: 'string'},
      {id: 'items', as: 'array'},
      {id: 'xLabel', as: 'string'},
      {id: 'yLabel', as: 'string'},
      {id: 'sizeLabel', as: 'string'},
      {id: 'xCategories', as: 'array'},
      {id: 'yCategories', as: 'array'},
      {id: 'donut', as: 'boolean', type: 'boolean<common>'},
      {id: 'showValues', as: 'boolean', type: 'boolean<common>'},
      {id: 'valueFormat', as: 'string'},
      {id: 'xFormat', as: 'string'},
      {id: 'yFormat', as: 'string'},
      {id: 'highlight', type: 'viz-highlight<viz>'},
      {id: 'drill', type: 'viz-drill<viz>[]'},
      {id: 'width', as: 'number'},
      {id: 'height', as: 'number'}
    ],
    impl: comp({
      hFunc: (ctx, { react: { hh } }, { items, ...props }) => runtimeProps => hh(ctx,
        dsls.react['react-comp'][`${templateParam[0].toLowerCase()}${templateParam.slice(1)}Widget`],
        { ...props, data: items, ...runtimeProps })
    })
  })
)

  })
