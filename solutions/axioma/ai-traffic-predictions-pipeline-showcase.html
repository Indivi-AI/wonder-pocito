-- Keyword breakdown: GSC prompts joined to request traffic.
-- Both sides use UTC day for alignment.
-- The request side is read from requests_agg (hourly grain) and rebucketed
-- into UTC day before joining.
-- Synthetic Google AI Overview rows are unioned from SerpAPI citations,
-- which retain the keyword grain that requests_agg_daily has already lost.
--
-- To avoid fan-out, requests are pre-aggregated to
-- (tenant_id, day, domain, normalized_url, agent) before the GSC join,
-- dropping page_id from the grouping key. ARS sub-metrics
-- (page_id != 'ORIGIN') are pre-computed as separate columns.
-- GSC metrics are assigned to a single representative agent row per
-- (tenant, day, domain, prompt, normalized_url) to prevent inflation.
WITH prompt_pages AS (
  SELECT
    tenant_id,
    TIMESTAMP_TRUNC(day, DAY) AS day,
    domain,
    prompt,
    MIN(
      REGEXP_REPLACE(REGEXP_REPLACE(page, r'#.*$', ''), r'/$', '')
    ) AS page_url,
    LOWER(
      REGEXP_REPLACE(
        REGEXP_REPLACE(
          REGEXP_REPLACE(page, r'^https?://', ''),
          r'#.*$',
          ''
        ),
        r'/$',
        ''
      )
    ) AS normalized_url,
    SUM(impressions) AS gsc_impressions,
    SUM(clicks) AS gsc_clicks,
    SAFE_DIVIDE(
      SUM(CASE WHEN position IS NOT NULL THEN impressions * position END),
      NULLIF(SUM(CASE WHEN position IS NOT NULL THEN impressions END), 0)
    ) AS gsc_position
  FROM ${ref("google_search_console_daily_normalized")}
  WHERE TRUE
    ${BACKFILL_START_DATE ? `AND day >= TIMESTAMP('${BACKFILL_START_DATE}')` : when(incremental(), `AND day >= TIMESTAMP(DATE_SUB(CURRENT_DATE(), INTERVAL ${GSC_LOOKBACK_DAYS} DAY))`)}
    ${devBronzeGuard("day")}
    ${BACKFILL_TENANT_ID ? `AND tenant_id = '${BACKFILL_TENANT_ID}'` : ""}
  GROUP BY tenant_id, day, domain, prompt, normalized_url
),

    ${devBronzeGuard("time")}
),

dedup AS (
  -- Dedup to one row per grain, picking a representative scan for the scalar fields
  -- (row_hash tie-break for determinism); URLs are unioned across all scans separately
  -- in exploded, so URLs from other scans in the grain aren't dropped.
  SELECT
    gsc_source_day,
    tenant_id,
    domain,
    agent,
    country,
    query,
    ARRAY_AGG(
      STRUCT(provider, answer_text, query_text)
      ORDER BY (CASE WHEN provider = 'serpapi' THEN 0 ELSE 1 END), time DESC, row_hash
      LIMIT 1
    )[OFFSET(0)] AS pick,
    -- MAX(time) keeps serpapi_citations_daily's derived `day` stable in the MERGE as
    -- rows age out of the window (AXI-3829).
    MAX(time) AS scanned_at
  FROM scan_rows
  GROUP BY gsc_source_day, tenant_id, domain, agent, country, query
),

exploded AS (
  -- Explode every scan in the grain (not just the pick) so URLs are unioned across
  -- re-scans/providers; per_url dedups by url. cited = citations/links_attached,
  -- sourced-only = references-only (AXI-3722). normalize() emits `url` on
  -- citations/links_attached; legacy rows use `link`, so accept both. references
  -- always uses `link`.
  SELECT gsc_source_day, tenant_id, domain, agent, country, query,
         COALESCE(JSON_VALUE(la, '$.url'), JSON_VALUE(la, '$.link')) AS url, TRUE AS in_citation
  FROM scan_rows, UNNEST(JSON_QUERY_ARRAY(links_attached)) AS la
  WHERE COALESCE(JSON_VALUE(la, '$.url'), JSON_VALUE(la, '$.link')) IS NOT NULL
  UNION ALL
  SELECT gsc_source_day, tenant_id, domain, agent, country, query,
         COALESCE(JSON_VALUE(ci, '$.url'), JSON_VALUE(ci, '$.link')) AS url, TRUE AS in_citation
  FROM scan_rows, UNNEST(JSON_QUERY_ARRAY(citations)) AS ci
  WHERE COALESCE(JSON_VALUE(ci, '$.url'), JSON_VALUE(ci, '$.link')) IS NOT NULL
  UNION ALL
  SELECT gsc_source_day, tenant_id, domain, agent, country, query,
         JSON_VALUE(r, '$.link') AS url, FALSE AS in_citation
  FROM scan_rows, UNNEST(JSON_QUERY_ARRAY(`references`)) AS r
  WHERE JSON_VALUE(r, '$.link') IS NOT NULL
),

per_url AS (
  SELECT
    gsc_source_day, tenant_id, domain, agent, country, query,
    url,
    LOWER(REGEXP_EXTRACT(url, r'^https?://([^/]+)')) AS cited_domain,
    LOGICAL_OR(in_citation) AS is_citation
  FROM exploded
  WHERE LENGTH(url) <= ${MAX_URL_LENGTH}
  GROUP BY gsc_source_day, tenant_id, domain, agent, country, query, url
),

url_agg AS (
  SELECT
    gsc_source_day, tenant_id, domain, agent, country, query,
    -- If a (malformed) grain exceeds the cap, keep the tenant's own domain
    -- first, then cited URLs, before the url tie-break — so the cap never drops
    -- owned citations that serpapi_citations_daily needs for GSC attribution.
    ARRAY_AGG(STRUCT(url, cited_domain, is_citation)
      ORDER BY
        (cited_domain = domain OR ENDS_WITH(cited_domain, CONCAT('.', domain))) DESC,
        is_citation DESC,
        url
      LIMIT ${MAX_URLS_PER_ANSWER}) AS urls
  FROM per_url
  GROUP BY gsc_source_day, tenant_id, domain, agent, country, query
)

SELECT
  a.gsc_source_day,
  a.tenant_id,
  a.domain,
  a.agent,
  a.country,
  a.query,
  a.pick.answer_text AS answer_text,
  a.pick.provider AS provider,
  a.pick.query_text AS query_text,
  a.scanned_at,
  u.urls
FROM dedup AS a
LEFT JOIN url_agg AS u
  USING (gsc_source_day, tenant_id, domain, agent, country, query)
